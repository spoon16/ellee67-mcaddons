 def test_unpatched_structure_matches_native_reference(self):
  new=read(A);reference=read(REF)
  for name,j in reference['animations'].items():

 def test_tp_actions_remain_nonconstant_without_engine_attachable_variables(self):
  e=env()

   if key.startswith('v.fp_melee_spear_') or key.startswith('v.tp_melee_spear_'):e.pop(key)
  for name in read(A)['animations']:
   if 'first_person' in name:continue # Now held constant; direct FP mesh has separate movement tests.
   channel='rotation'
   bone='rightitem' if 'first_person' in name else 'rightarm'
   expression=read(A)['animations'][name]['bones'][bone][channel][0]
   e0=e|{'v.attack_time':0,'q.main_hand_item_use_duration':0}
   e1=e|{'v.attack_time':.3,'q.main_hand_item_use_duration':20}
   self.assertNotEqual(evaluate(expression,e0),evaluate(expression,e1))
