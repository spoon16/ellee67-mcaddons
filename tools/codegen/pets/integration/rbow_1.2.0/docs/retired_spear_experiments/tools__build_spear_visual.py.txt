#!/usr/bin/env python3
"""Bound compact spear made from the EXISTING 32px inventory sprite.

The 1.1.6 player/rightitem curves did not animate the user's first-person
ordinary item renderer. This explicit mesh is an attachable and its own bone
now receives the action. It is NOT the retired texture_meshes geometry.

Coordinates are model units (16/block). A half-unit per source texel preserves
normal 16-unit icon scale. Grip is modeled at [0,24,0] to account for the bound
item-bone -24 Y convention. FP pose is fitted against the standard player base
and melee-spear hold poses; this fitting is NOT a Minecraft-client test.

No raster artwork, item statistics, attack scripts or vanilla entities are edited.
"""
from __future__ import annotations
import hashlib, json, math
from pathlib import Path
from PIL import Image

ROOT=Path(__file__).resolve().parents[1]
RP=ROOT/'resource_pack'
NS='elleedog.rbow_spear'
ANCHOR=[0.0,24.0,0.0]
GRIP_PIXEL=[10.5,20.5]
PIXEL=0.5
THICKNESS=0.5
IS_PLAYER="q.is_owner_identifier_any('minecraft:player')"
IS_MAIN="c.item_slot == 'main_hand'"

def save(path, obj):
    path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(json.dumps(obj,indent=2)+'\n')

def build():
    sprite=RP/'textures/items/rbow_spear.png'
    image=Image.open(sprite).convert('RGBA')
    if image.size != (32,32):raise ValueError('Pinned spear must be 32x32.')
    item=json.loads((ROOT/'behavior_pack/items/rbow_spear.json').read_text())['minecraft:item']['components']
    kinetic=item['minecraft:kinetic_weapon']
    delay=max(.05,kinetic['delay']/20)
    sway_start=(kinetic['delay']+kinetic['dismount_conditions']['max_duration'])/20
    finish=(kinetic['delay']+kinetic['damage_conditions']['max_duration'])/20
    # One shallow cube per OPAQUE pixel; no rectangle across transparent space.
    # Same source texel on every exterior face. Avoid internal coincident faces.
    def occupied(x,y):return 0<=x<32 and 0<=y<32 and image.getpixel((x,y))[3]>0
    cubes=[]
    for y in range(32):
        for x in range(32):
            if not occupied(x,y):continue
            if image.getpixel((x,y))[3]!=255:raise ValueError('Unexpected fractional alpha.')
            faces=['north','south']
            if not occupied(x-1,y):faces.append('west')
            if not occupied(x+1,y):faces.append('east')
            if not occupied(x,y-1):faces.append('up')
            if not occupied(x,y+1):faces.append('down')
            cubes.append({
                'origin':[round((x-GRIP_PIXEL[0])*PIXEL,5),
                          round(24+(GRIP_PIXEL[1]-y-1)*PIXEL,5),-THICKNESS/2],
                'size':[PIXEL,PIXEL,THICKNESS],
                'uv':{face:{'uv':[x+.01,y+.01],'uv_size':[.98,.98]} for face in faces}
            })
    geo={'format_version':'1.16.0','minecraft:geometry':[{
        'description':{'identifier':'geometry.'+NS+'.compact', 'texture_width':32,'texture_height':32,
                       'visible_bounds_width':3,'visible_bounds_height':3,'visible_bounds_offset':[0,1.5,0]},
        'bones':[
            {'name':'rbow_grip','pivot':ANCHOR,'binding':'q.item_slot_to_bone_name(c.item_slot)'},
            {'name':'rbow_axis','parent':'rbow_grip','pivot':ANCHOR},
            {'name':'rbow_pixels','parent':'rbow_axis','pivot':ANCHOR,'rotation':[0,0,45],'cubes':cubes}
        ]}]}
    save(RP/'models/entity/rbow_spear_compact.geo.json',geo)
    prep=[
        f'v.rb_player = {IS_PLAYER};',
        f'v.rb_main = {IS_MAIN};',
        'v.rb_attack = (v.rb_player && v.rb_main) ? math.clamp((c.owning_entity->v.attack_time ?? 0.0), 0.0, 1.0) : 0.0;',
        'v.rb_using = (v.rb_player && v.rb_main) ? ((c.owning_entity->v.item_use_normalized ?? 0.0) > 0.0) : 0.0;',
        # Query delta_time is in seconds. Avoid guessing native item-use query units.
        f'v.rb_use_seconds = v.rb_using ? math.min({finish+1}, (v.rb_use_seconds ?? 0.0) + math.max(0.0, q.delta_time)) : 0.0;',
        f'v.rb_raise = math.clamp(v.rb_use_seconds / {delay}, 0.0, 1.0);',
        f'v.rb_lower = math.clamp((v.rb_use_seconds - {finish-.6}) / 0.6, 0.0, 1.0);',
        'v.rb_ready = v.rb_raise * (1.0 - v.rb_lower);',
        f'v.rb_sway = math.sin(q.life_time * 150.0) * math.clamp((v.rb_use_seconds - {sway_start}) / 1.0, 0.0, 1.0) * v.rb_ready;',
        'v.rb_jab = v.rb_using ? 0.0 : math.sin(math.pow(v.rb_attack, 0.55) * 180.0);'
    ]
    anims={
        # Idle is fitted to the STANDARD vanilla player arm / rightitem frame.
        # The root stays centered on the grip. The shaft may slide by at most
        # 1.5 units through the hand during a jab, never 24/27 units away.
        'animation.'+NS+'.first_person':{
            'loop':True,
            'bones':{
                'rbow_grip':{
                    'rotation':['94.1796 - 57.0 * v.rb_ready - 72.0 * v.rb_jab',21.0682,'142.7284 + 1.2 * v.rb_sway'],
                    'position':[0,0,0]
                },
                'rbow_axis':{'position':[0,'1.5 * v.rb_jab',0]}
            }
        },
        'animation.'+NS+'.third_person':{
            'loop':True,'bones':{
                # TP limb/action curves from 1.1.6 still move the owner's arm.
                # No separate shaft displacement on other players / armor stands.
                'rbow_grip':{'rotation':[0,0,0],'position':[0,0,0]},
                'rbow_axis':{'position':[0,0,0]}
            }
        }
    }
    save(RP/'animations/rbow_spear_direct.animation.json',{'format_version':'1.10.0','animations':anims})
    save(RP/'attachables/rbow_spear_compact.json',{
        'format_version':'1.20.30',
        'minecraft:attachable':{'description':{
            'identifier':'elleedog:rbow_spear',
            'materials':{'default':'entity_alphatest','enchanted':'entity_alphatest_glint'},
            'textures':{'default':'textures/items/rbow_spear','enchanted':'textures/misc/enchanted_item_glint'},
            'geometry':{'default':'geometry.'+NS+'.compact'},
            'animations':{'fp':'animation.'+NS+'.first_person','tp':'animation.'+NS+'.third_person'},
            'scripts':{
                'initialize':['v.rb_use_seconds = 0.0;'],
                'pre_animation':prep,
                'animate':[{'fp':'c.is_first_person'},{'tp':'!c.is_first_person'}]
            },
            'render_controllers':['controller.render.item_default']
        }}
    })
    save(ROOT/'docs/v1.1.7_spear_visual.json',{
        'version':[1,1,7],'engine_tests_run':0,'client_render_tests_run':0,'realm_tests_run':0,
        'baseline_user_report':'1.1.5 size/placement worked; 1.1.6 first-person attack/use sounded but remained upright.',
        'strategy':'Explicit compact sprite extrusion bound to the hand, with own FP action bone. No texture_meshes.',
        'sprite_sha256':hashlib.sha256(sprite.read_bytes()).hexdigest(),
        'opaque_pixels':len(cubes),'model_units_per_source_pixel':PIXEL,'depth_model_units':THICKNESS,
        'grip_pixel_center':GRIP_PIXEL,'binding_anchor_model_units':ANCHOR,
        'fp_idle_euler_degrees':[94.1796,21.0682,142.7284],
        'fp_ready_pitch_delta_degrees':-57,'fp_jab_pitch_delta_degrees':-72,
        'max_shaft_slide_model_units':1.5,'charge_raise_seconds':delay,'charge_finish_seconds':finish,
        'pose_fit_assumptions':{'player_arm_xyz':[95,-45,115],'player_rightitem_xyz':[45,55,-12.5],
                               'target_camera_xyz':[-8,-20,-12],
                               'rotation_order':'Rz * Ry * Rx; numeric fit, not a client-calibrated pose'},
        'known_risks':['Compact attachable replaces the held sprite only; actual POV/hand fit still needs client testing.',
                       'Other player-animation packs or nonstandard character geometries can change the assumed parent frame.',
                       'Visible motion depends on native public owner attack_time and item_use_normalized being supplied.'],
        'changed_features':['FP animation follows visible model bones; no dependency on engine-only sprite movement.',
                            'TP attack/use arm curves retained; item translation kept at native baseline.',
                            'Inventory sprite, dropped items and gameplay remain unchanged.']
    })
    print(f'Built compact hand-bound spear: {len(cubes)} pixel cubes, existing texture; FP ready/jab bones wired.')

if __name__=='__main__':build()
